/* ══════════════════════════════════════════════════════════════
   RENDER — EXPLORER
   ══════════════════════════════════════════════════════════════ */
function renderExplorer(){
  const fileCount = Object.values(TREE).reduce((n,m)=>n+m.files.length,0);
  const dirCount = Object.values(TREE).reduce((n,m)=>n+m.dirs.length,0);

  let html = `<div class="exp-head"><span>explorer</span><span class="badge">${Object.keys(MODULES).length}</span></div>`;

  for (const [modId, mod] of Object.entries(MODULES)){
    const tree = TREE[modId];
    const isOpen = state.openDirs[modId];
    const active = state.module === modId;

    html += `<div class="tree-group">`;
    html += `
      <div class="tree-item ${active?'active':''}" data-mod="${modId}">
        <span class="ic" style="transform:rotate(${isOpen?90:0}deg); color:${mod.color}">▸</span>
        <span class="name">${mod.label}</span>
        <span class="meta">${tree.files.length}f · ${tree.dirs.length}d</span>
      </div>
    `;

    html += `<div class="tree-sub ${isOpen?'open':''}">`;

    // dirs
    for (const d of tree.dirs){
      const dirFiles = tree.files.filter(f => f.id.startsWith(`${modId}/${d}/`));
      html += `
        <div class="tree-item" data-dir="${modId}/${d}">
          <span class="ic">▸</span>
          <span class="name">${d}</span>
          <span class="meta">${dirFiles.length}</span>
        </div>
      `;
      html += `<div class="tree-sub">`;
      for (const f of dirFiles){
        html += fileItemHtml(f);
      }
      html += `</div>`;
    }

    // root files
    for (const f of tree.files){
      const isNested = tree.dirs.some(d => f.id.startsWith(`${modId}/${d}/`));
      if (!isNested) html += fileItemHtml(f);
    }

    html += `</div></div>`;
  }

  $('explorer').innerHTML = html;

  // wire
  $('explorer').querySelectorAll('[data-mod]').forEach(el => {
    el.onclick = (e) => {
      e.stopPropagation();
      const mod = el.dataset.mod;
      if (el.classList.contains('tree-item') && el.querySelector('.ic')){
        // toggle dir if it has children
        if (state.module === mod && el.parentElement.classList.contains('tree-sub') === false){
          state.openDirs[mod] = !state.openDirs[mod];
        } else {
          state.openDirs[mod] = true;
        }
      }
      selectModule(mod);
    };
  });

  $('explorer').querySelectorAll('[data-dir]').forEach(el => {
    el.onclick = (e) => {
      e.stopPropagation();
      const sub = el.nextElementSibling;
      if (sub) sub.classList.toggle('open');
      el.querySelector('.ic').style.transform = sub.classList.contains('open') ? 'rotate(90deg)' : 'rotate(0)';
    };
  });

  $('explorer').querySelectorAll('[data-file]').forEach(el => {
    el.onclick = (e) => {
      e.stopPropagation();
      selectFile(el.dataset.file);
    };
  });
}

function fileItemHtml(f){
  const active = state.file === f.id;
  return `
    <div class="tree-item ${active?'active':''}" data-file="${f.id}">
      <span class="dot-state" style="background:${stateColor(f.state)}"></span>
      <span class="fname">${f.name}</span>
      <span class="meta">${f.size}</span>
    </div>
  `;
}

function firstMeaningfulLine(content){
  const lines = content.split('\n').map(l=>l.trim()).filter(l => l && !l.startsWith('//') && !l.startsWith('#'));
  return esc((lines[0] || '').slice(0, 120)) + (lines[0] && lines[0].length > 120 ? '…' : '');
}

function highlight(content, lang){
  let html = esc(content);
  // comments first
  if (lang === 'typescript'){
    html = html.replace(/(\/\/[^\n]*)/g, '<span class="cmt">$1</span>');
    html = html.replace(/\b(export|interface|function|const|type|return|import|from)\b/g, '<span class="kw">$1</span>');
    html = html.replace(/"([^"]*)"/g, '<span class="str">"$1"</span>');
    html = html.replace(/\b(\d+)\b/g, '<span class="num">$1</span>');
  } else if (lang === 'markdown'){
    html = html.replace(/(^#{1,6} [^\n]*)/gm, '<span class="kw">$1</span>');
    html = html.replace(/(^[-*] )/gm, '<span class="cmt">$1</span>');
  }
  return html;
}

const COMMANDS = [
  { id:'nav-workspace', label:'Go to Workspace', sub:'module', kbd:'⌘1', act:()=>selectModule('workspace') },
  { id:'nav-kernel', label:'Go to Kernel', sub:'module', kbd:'⌘2', act:()=>selectModule('kernel') },

  { id:'agent-toggle', label:'Toggle Hermes Agent', sub:'⌘⇧A', kbd:'', act:()=>toggleAgent() },
  { id:'mint', label:'Mint current file to The Roll', sub:'action', kbd:'', act:()=>toast('Mint queued · anchor signature required', '◈') },
];

function allFiles(){
  const out = [];
  for (const [mod, tree] of Object.entries(TREE)){
    for (const f of tree.files){
      out.push({ id:f.id, label:f.name, sub:f.id, kbd:'', act:()=>selectFile(f.id) });
    }
  }
  return out;
}

function filteredPalette(){
  const q = state.paletteQuery.toLowerCase().trim();
  const cmds = COMMANDS;
  const files = allFiles();
  if (!q) return { cmds: cmds.slice(0,7), files: files.slice(0,5) };
  return {
    cmds: cmds.filter(c => c.label.toLowerCase().includes(q) || c.sub.toLowerCase().includes(q)),
    files: files.filter(f => f.label.toLowerCase().includes(q) || f.sub.toLowerCase().includes(q)),
  };
}

function flatPalette(){
  const { cmds, files } = filteredPalette();
  return [...cmds, ...files];
}

function renderPalette(){
  const { cmds, files } = filteredPalette();
  const flat = flatPalette();
  if (state.paletteIdx >= flat.length) state.paletteIdx = Math.max(0, flat.length-1);

  let html = '';
  if (cmds.length){
    html += `<div class="palette-section">commands</div>`;
    cmds.forEach((c, i) => {
      html += paletteItemHtml(c, i, '⌘');
    });
  }
  if (files.length){
    html += `<div class="palette-section">files</div>`;
    files.forEach((c, i) => {
      html += paletteItemHtml(c, cmds.length + i, '▤');
    });
  }
  if (!cmds.length && !files.length){
    html += `<div class="palette-section" style="text-align:center; padding:32px">No matches</div>`;
  }
  $('paletteList').innerHTML = html;

  $('paletteList').querySelectorAll('[data-pidx]').forEach(el => {
    el.onclick = () => {
      const idx = parseInt(el.dataset.pidx);
      runPaletteIndex(idx);
    };
    el.onmouseenter = () => {
      state.paletteIdx = parseInt(el.dataset.pidx);
      updatePaletteActive();
    };
  });
  updatePaletteActive();
}

function paletteItemHtml(item, idx, icon){
  const active = idx === state.paletteIdx;
  return `
    <div class="palette-item ${active?'active':''}" data-pidx="${idx}">
      <div class="ic">${icon}</div>
      <div class="label">
        <div>${esc(item.label)}</div>
        <div class="sub">${esc(item.sub)}</div>
      </div>
      ${item.kbd ? `<span class="kbd">${item.kbd}</span>` : ''}
    </div>
  `;
}

function updatePaletteActive(){
  $('paletteList').querySelectorAll('[data-pidx]').forEach(el => {
    el.classList.toggle('active', parseInt(el.dataset.pidx) === state.paletteIdx);
  });
  const active = $('paletteList').querySelector('.palette-item.active');
  if (active) active.scrollIntoView({ block:'nearest' });
}

function runPaletteIndex(idx){
  const flat = flatPalette();
  const item = flat[idx];
  if (!item) return;
  closePalette();
  item.act();
}

function openPalette(){
  state.paletteOpen = true;
  state.paletteQuery = '';
  state.paletteIdx = 0;
  $('paletteOverlay').classList.add('open');
  $('paletteInput').value = '';
  $('paletteInput').focus();
  renderPalette();
}

function closePalette(){
  state.paletteOpen = false;
  $('paletteOverlay').classList.remove('open');
}

$('searchBtn').onclick = openPalette;
$('paletteOverlay').onclick = closePalette;

$('paletteInput').addEventListener('input', e => {
  state.paletteQuery = e.target.value;
  state.paletteIdx = 0;
  renderPalette();
});

$('paletteInput').addEventListener('keydown', e => {
  const flat = flatPalette();
  if (e.key === 'ArrowDown'){ e.preventDefault(); state.paletteIdx = Math.min(flat.length-1, state.paletteIdx+1); updatePaletteActive(); }
  else if (e.key === 'ArrowUp'){ e.preventDefault(); state.paletteIdx = Math.max(0, state.paletteIdx-1); updatePaletteActive(); }
  else if (e.key === 'Enter'){ e.preventDefault(); runPaletteIndex(state.paletteIdx); }
  else if (e.key === 'Escape'){ e.preventDefault(); closePalette(); }
});

const SCRIPTED = [
  "Reading tree. I see {file} selected. Want me to audit the BOM against the import quote, or check the anchor signature chain?",
  "Note: {file} is in state {state}. I can propose the next evidence step — either fabricate a prototype or schedule an audit.",

function agentSay(msg, who='hermes'){
  const el = document.createElement('div');
  el.className = `msg ${who==='user'?'user':''}`;
  el.innerHTML = `<span class="p">${who} ›</span> ${msg}`;
  $('agentBody').appendChild(el);
  $('agentBody').scrollTop = $('agentBody').scrollHeight;
}

function agentRespond(){
  const f = state.file ? findFile(state.file) : null;
  const tmpl = SCRIPTED[Math.floor(Math.random()*SCRIPTED.length)];
  const reply = tmpl
    .replace('{file}', f ? f.name : 'no file')
    .replace('{state}', f ? f.state : 'unknown');
  setTimeout(()=>agentSay(reply), 500 + Math.random()*600);
}

$('agentInput').addEventListener('keydown', e => {
  if (e.key === 'Enter'){
    const v = e.target.value.trim();
    if (!v) return;
    agentSay(esc(v), 'user');
    e.target.value = '';
    agentRespond();
  }
});

/* ══════════════════════════════════════════════════════════════
   KEYBINDINGS
   ══════════════════════════════════════════════════════════════ */
document.addEventListener('keydown', e => {
  const meta = e.metaKey || e.ctrlKey;

  if (meta && e.key.toLowerCase() === 'k'){
    e.preventDefault();
    state.paletteOpen ? closePalette() : openPalette();
    return;
  }
  if (meta && e.key.toLowerCase() === 'p'){
    e.preventDefault();
    openPalette();
    return;
  }
  if (meta && e.shiftKey && e.key.toLowerCase() === 'a'){
    e.preventDefault();
    toggleAgent();
    return;
  }
  if (meta && e.key >= '1' && e.key <= '6'){
    e.preventDefault();
    const mods = Object.keys(MODULES);
    const idx = parseInt(e.key) - 1;
    if (mods[idx]) selectModule(mods[idx]);
    return;
  }
  if (e.key === 'Escape'){
    if (state.paletteOpen) closePalette();
  }
});
